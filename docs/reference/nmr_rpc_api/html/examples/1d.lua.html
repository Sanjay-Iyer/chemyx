1d.lua

---------------------------
-- NMReady 1D experiment --
---------------------------

--------------------------
-- Required api modules --
--------------------------
local api = require("api")
local savehelper = api.operations.callbacks.savehelper
local designerui = api.ui.designer
local transientCallbackHelper =
    api.operations.callbacks.transientcallbackhelper
local superoned = api.operations.superoned
local common = api.operations.common

----------------------------------------
-- Define the GUI and get dummy scans --
----------------------------------------
local pulseAngle = designerui.number("Pulse Angle", 2, 90, 0, 360)
local dummyScans = core:dummyScans()

-------------------------------------------------------
-- Create variable for data storage and register it, --
-- so data is saved if the save button is pressed  --
-------------------------------------------------------
local dataset1d = newDataSet1D("1D DataSet")
savehelper.saveDataOnSaveButton(nil, "1D Data", dataset1d)

---------------------------------------------------------
-- Create displays and pages where data will be viewed --
---------------------------------------------------------
local page = DisplayPage{name = "Data Page"}
local display = Display1D(page, {name = "Data Display",
    datasetName = "1D DataSet", grid = "off"})

--------------------
-- Pulse sequence --
--------------------
local chan1 = core:channelConfigSignal()
local p0 = chan1:pulseTimeForAngle(pulseAngle)

local p = PulseSequence()
p:add(newPulseSingle())

p:setPulseTime(0, p0)

-----------------
-- Phase cycle --
-----------------
local phaseTable = PhaseTable()
phaseTable:addEntries({
    {0, 0},
    {180, 180}
})

phaseTable:setAccumulates(true)

---------------------------------
-- Transient callback function --
---------------------------------
local function transientCallback(currTransient, totalTransients)
    local progress = (100.0 * currTransient) / totalTransients
    local msg = string.format("1D: Running scan #%d of %d",
        currTransient, totalTransients)
    setProgressMessage(msg)
    setStatusMessage(msg)
    setProgress(progress)

    -- Update the display
    dataset1d:set(core:dataSetSignal())
    updateDisplay(dataset1d)
end

transientCallbackHelper.registerExperimentCallback(transientCallback)

-------------------------------------
-- The main body of the experiment --
-------------------------------------
core:usePulseSequence(p)
setPhaseTable(phaseTable)
setCalculateIntermediateResults(true)

setProgress(0)

common.runDummyScans{dummyScans=dummyScans}

superoned.acquireSingle1D()

setProgress(100)
setProgressMessage("Complete")

dataset1d:set(core:dataSetSignal())

updateDisplay(dataset1d)
