cosy.lua

--------------------------------------------------------------
-- This is an example implementation of the COSY experiment --
--------------------------------------------------------------

-------------------------------------------------
-- Start by declaring the required api modules --
-------------------------------------------------
local api = require("api")
local standard = api.helpers.standard
local savehelper = api.operations.callbacks.savehelper
local common = api.operations.common
local designerui = api.ui.designer

---------------------------------------
-- Then define a GUI for this script --
---------------------------------------
local t1NumPoints = designerui.number("t1 Number of Points", 0, 64, 2, 2048)
local t1StartTimeMs = designerui.number("t1 Start Time (ms)", 2, 0.1, 0, 30000)
local f1SpectralWidthPpm = designerui.number("f1 Spectral Width (ppm)", 0, 12, 4, 1000)
local second90PulseUs = designerui.number("Second 90 Pulse (us)", 2, 30, 0, 1000)

-- Convert some values for later use
local secondPulse90Seconds = standard.microsecondsToSeconds(second90PulseUs)
local f1SpectralWidthHz = common.ppmToHertz(f1SpectralWidthPpm)
local t1StepSizeMs = standard.secondsToMilliseconds(common.hertzToSeconds(f1SpectralWidthHz))
local dummyScans = core:dummyScans()

-------------------------------------------------
-- Setup Phase table and force number of scans --
-------------------------------------------------
local phaseTableEntries = {
    { 0.0, 270.0, 0.0 },
    { 0.0, 180.0, 180.0 },
    { 0.0, 90.0, 0.0 },
    { 0.0, 0.0, 180.0 }
}
local roundedNumberOfScans = standard.roundUpToNearestStrictlyPositiveMultiple(
        core:numScans(),
        #phaseTableEntries
)
core:setNumScans(roundedNumberOfScans)
local phaseTable = PhaseTable()
phaseTable:addEntries(phaseTableEntries)
phaseTable:setAccumulates(true)
setPhaseTable(phaseTable)

-------------------------------------------------------------------
-- Create variables where experiment data will be stored,        --
-- and register them to be saved if the save button is pressed --
-------------------------------------------------------------------
local results = newResults2D("COSY Results")
results:setSpectralWidthF1Ppm(f1SpectralWidthPpm)
results:setSwapHalvesF1(true)
results:setMirrorF1(true)
results:setSwapHalvesF2(true)
results:setMirrorF2(false)
savehelper.saveDataOnSaveButton(nil, "COSY", results)

---------------------------------------------------------
-- Create displays and pages where data will be viewed --
---------------------------------------------------------
local page = DisplayPage{ name = "COSY Results" }
local display = Display2D(page, { name = "COSY Results", datasetName = "COSY Results", mode = "magnitude" })

-- Wait for the length of an interscan delay before the first acquisition
setPreScanDelay(core:channelConfigSignal():interscanTime())

---------------------------
-- Run reference Scans.  --
---------------------------
setProgressMessage("Acquiring Reference")
common.runReferenceScan{
    results=results,
    xReference=true,
    yReference=true,
    defaultPulseSequence=true,
    numScans=roundedNumberOfScans
}
-------------------------------------------------------------------------------------
-- Make helper functions to create a pulse sequence, calculate tau, and callback   --
-- functionality for the COSY experiment and set some values in the core           --
-------------------------------------------------------------------------------------
local function tauPerAcquisition(acquisition)
    return standard.millisecondsToSeconds(t1StepSizeMs * acquisition + t1StartTimeMs)
end

local function createPulseSequence(acquisition)
    local p = PulseSequence(core:channelConfigSignal())
    p:add(newPulseSingle(core:channelConfigSignal()))
    p:add(newPulseSingle(core:channelConfigSignal()))
    p:setPulseTime(0, core:channelConfigSignal():get90PulseSeconds())
    p:setPulseTime(1, secondPulse90Seconds)
    p:setInterPulseTime(0, tauPerAcquisition(acquisition))
    core:usePulseSequence(p)
end
local function newResultAddedCallback()
    -- "acquisition" is zero based.
    local acquisition = results:numDataRows() - 1
    setProgress(100 * acquisition / (t1NumPoints - 1))
    setProgressMessage("Running point " .. tostring(acquisition+1) .. " of " .. tostring(t1NumPoints))
    results:process2D()
    updateDisplay(results)
end

-- Run the main acquisitions based on the above functions
common.runMulti1DScans{
    setupMethod=createPulseSequence,
    yMethod=tauPerAcquisition,
    results=results,
    numPoints=t1NumPoints,
    dummyScans=dummyScans,
    newResultAddedCallback=newResultAddedCallback,
}

-- Process and update the display one last time
results:process2D()
updateDisplay(results)

setProgress(100)
setProgressMessage("Complete")
