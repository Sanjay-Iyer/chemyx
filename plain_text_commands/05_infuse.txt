# What it does : configures the pump, sets a POSITIVE volume, and starts

# Expected echoes (one per line):

#       diameter = 4.5
#       rate = 1.0
#       volume = 0.5          <-- positive = infuse
#       pump start
# SAFETY: the last line ("start") makes the pump MOVE. Make sure the
#   syringe, tubing and collection vessel are ready first.
# Channel (4000X): prefix each line with the channel number to target one
#   drive, e.g.  "2 set rate 1.0" ... "2 start".
# REMINDER: press Enter (carriage return, \r) after EVERY line.
# NOTE: lines starting with # are notes for YOU — do NOT type them.
# ------------------- SEND THE LINES BELOW, IN ORDER ------------------
set units 0
set diameter 4.5
set rate 1.0
set volume 0.5
start
